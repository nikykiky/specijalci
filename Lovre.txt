CREATE TABLE uredaj (
	id_uredaj INT PRIMARY KEY AUTO_INCREMENT,
    naziv VARCHAR(100) NOT NULL,
    tip VARCHAR(50),
    lokacija VARCHAR(100)
);


CREATE TABLE prijava_kvara (
	id_prijave INT PRIMARY KEY AUTO_INCREMENT,
    id_uredaja INT NOT NULL,
    datum_prijave DATE NOT NULL,
    opis VARCHAR(255)
    
    FOREIGN KEY (id_uredaja)REFERENCES uredaj(id_uredaja)
);



CREATE TABLE djelatnik (
	id_djelatnik INT PRIMARY KEY AUTO_INCREMENT,
    ime VARCHAR(25),
    prezime VARCHAR(25),
    radno_mjesto VARCHAR(50),
    telefon INT(12) NOT NULL
);
CREATE TABLE intervencija (
	id_intervencije INT PRIMARY KEY AUTO_INCREMENT,
    id_prijave INT NOT NULL,
    id_djelatnika INT NOT NULL,
    datum_intervencije DATE NOT NULL,
    trajanje INT NOT NULL,
    ishod VARCHAR(50),
    
    FOREIGN KEY (id_prijave)REFERENCES prijava_kvara(id_prijave),
    FOREIGN KEY (id_djelatnika)REFERENCES djelatnik(id_djelatnik)
);



FOREIGN KEY (id_djelatnika)REFERENCES djelatnik(id_djelatnik)
